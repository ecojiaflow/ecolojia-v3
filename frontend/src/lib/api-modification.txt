export interface SearchFilters {
  brands?: string[];
  categories?: string[];
  productType?: string;
  minScore?: number;
  maxScore?: number;
}

// Fonction helper pour construire les query params
function buildQueryParams(query: string, filters: SearchFilters = {}): string {
  const params = new URLSearchParams();
  params.append('q', query);
  
  if (filters.brands?.length) {
    filters.brands.forEach(brand => params.append('brands', brand));
  }
  if (filters.categories?.length) {
    filters.categories.forEach(cat => params.append('categories', cat));
  }
  if (filters.productType) {
    params.append('productType', filters.productType);
  }
  if (filters.minScore !== undefined) {
    params.append('minScore', filters.minScore.toString());
  }
  if (filters.maxScore !== undefined) {
    params.append('maxScore', filters.maxScore.toString());
  }
  
  return params.toString();
}

// Modifier la fonction searchProducts existante
export async function searchProducts(query: string, filters: SearchFilters = {}): Promise<SearchResult> {
  const q = query.trim();
  if (!q) return { items: [], source: "local" };

  const queryParams = buildQueryParams(q, filters);

  // 1) Tentative Algolia (unifiée)
  try {
    const data = await fetchJSON(`${API_BASE}/api/algolia/search?${queryParams}`);
    const items = normalizeItemsFromAlgolia(data);
    if (items.length > 0) return { items, source: "algolia" };
  } catch {
    // ignore -> fallback
  }

  // 2) Fallback local
  try {
    const data = await fetchJSON(`${API_BASE}/api/products?${queryParams}`);
    const items = normalizeItemsFromLocal(data);
    return { items, source: "local" };
  } catch {
    return { items: [], source: "local" };
  }
}
